          <p class="editorial-prose">
            When an inventor copies an evolutionary mechanism, who owns the design? Under Section 3(c) of the Indian Patents Act, 1970, and US 35 U.S.C. § 101 doctrine, natural organisms and principles of nature cannot be patented. However, non-obvious synthetic adaptations are patentable. This creates a critical jurisprudential boundary: protecting inventive engineering without creating corporate monopolies over evolutionary solutions.
          </p>

            <div>• <strong>The Public Domain:</strong> Ensuring sustainable biomimetic designs remain accessible rather than tied up in predatory patent litigation.</div>
          </div>

    /* State Controller */
    let currentSlide = 0;
    let prompterLang = 'en';
    let outroHasTriggered = false;
    let audioCtxInstance = null;

    /* DOM References */
    const slides = document.querySelectorAll('.slide');
    const totalSlides = slides.length;
    const progressFill = document.getElementById('progress-fill');
    const currentSlideNum = document.getElementById('current-slide-num');
    const prevBtn = document.getElementById('prev-btn');
    const nextBtn = document.getElementById('next-btn');
    const drawer = document.getElementById('script-drawer');
    const drawerClose = document.getElementById('drawer-close');
    const drawerHeading = document.getElementById('drawer-heading');
    const drawerText = document.getElementById('drawer-text');
    const prompterLangBtn = document.getElementById('prompter-lang-btn');
    const outroAudio = document.getElementById('outro-sound');

    /* Lightbox References */
    const lightboxModal = document.getElementById('lightbox-modal');
    const lightboxImg = document.getElementById('lightbox-img');
    const lightboxCaption = document.getElementById('lightbox-caption');

    /* Chime Sound (Plays once on Slide 8) */
    function playSynthesizedMajorChime() {
      try {
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        if (!audioCtxInstance) audioCtxInstance = new AudioCtx();
        if (audioCtxInstance.state === 'suspended') audioCtxInstance.resume();

        const freqs = [523.25, 659.25, 783.99, 1046.50];
        freqs.forEach((freq, idx) => {
          const osc = audioCtxInstance.createOscillator();
          const gain = audioCtxInstance.createGain();
          const t = audioCtxInstance.currentTime + idx * 0.08;

          osc.type = 'sine';
          osc.frequency.setValueAtTime(freq, t);
          gain.gain.setValueAtTime(0, t);
          gain.gain.linearRampToValueAtTime(0.12, t + 0.04);
          gain.gain.exponentialRampToValueAtTime(0.001, t + 1.5);

          osc.connect(gain);
          gain.connect(audioCtxInstance.destination);

          osc.start(t);
          osc.stop(t + 1.6);
        });
      } catch (err) {}
    }

    function triggerOutroSoundOnce() {
      if (outroHasTriggered) return;
      outroHasTriggered = true;

      if (outroAudio && outroAudio.src) {
        const p = outroAudio.play();
        if (p !== undefined) {
          p.catch(() => playSynthesizedMajorChime());
        }
      } else {
        playSynthesizedMajorChime();
      }
    }

    /* Update Teleprompter Content */
    function updatePrompter() {
      const cuePrefix = prompterLang === 'en' ? 'STAGE DIRECTIVE: ' : 'STAGE HIDAYAT: ';
      drawerHeading.textContent = prompterLang === 'en' ? 'REHEARSAL PROMPTER' : 'PROMPTER (HINGLISH)';
      prompterLangBtn.textContent = prompterLang === 'en' ? 'EN | हिं' : 'HI | EN';

      const note = SPEAKER_SCRIPTS[prompterLang][currentSlide];
      drawerText.innerHTML = `
        <div class="stage-directive-box">${cuePrefix}${note.cue}</div>
        <p>${note.script}</p>
      `;
    }

    /* Presentation Navigation */
    function updatePresentationUI() {
      slides.forEach((slide, idx) => {
        slide.classList.remove('active', 'previous');
        if (idx === currentSlide) {
          slide.classList.add('active');
        } else if (idx < currentSlide) {
          slide.classList.add('previous');
        }
      });

      const percentage = ((currentSlide + 1) / totalSlides) * 100;
      progressFill.style.width = `${percentage}%`;
      currentSlideNum.textContent = String(currentSlide + 1).padStart(2, '0');

      updatePrompter();

      if (currentSlide === totalSlides - 1 && !outroHasTriggered) {
        setTimeout(triggerOutroSoundOnce, 320);
      }
    }

    function nextSlide() {
      if (currentSlide < totalSlides - 1) {
        currentSlide++;
        updatePresentationUI();
      }
    }

    function prevSlide() {
      if (currentSlide > 0) {
        currentSlide--;
        updatePresentationUI();
      }
    }

    /* Lightbox Engine */
    document.querySelectorAll('[data-zoom]').forEach(card => {
      card.addEventListener('click', (e) => {
        e.stopPropagation();
        const src = card.getAttribute('data-zoom');
        const cap = card.getAttribute('data-caption') || 'Figure Inspection';
        const innerImg = card.querySelector('img');

        lightboxImg.src = src;
        lightboxImg.onerror = () => {
          if (innerImg) lightboxImg.src = innerImg.src;
        };
        lightboxCaption.textContent = cap;
        lightboxModal.classList.add('active');
      });
    });

    lightboxModal.addEventListener('click', () => {
      lightboxModal.classList.remove('active');
    });

    /* Keyboard Controller */
    window.addEventListener('keydown', (e) => {
      const activeTag = document.activeElement ? document.activeElement.tagName.toLowerCase() : '';
      if (activeTag === 'input' || activeTag === 'textarea' || document.activeElement.isContentEditable) return;

      if (e.key === 'Escape') {
        if (lightboxModal.classList.contains('active')) {
          lightboxModal.classList.remove('active');
          return;
        }
        if (drawer.classList.contains('open')) {
          drawer.classList.remove('open');
          return;
        }
      }

      if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'PageDown') {
        e.preventDefault();
        nextSlide();
      } else if (e.key === 'ArrowLeft' || e.key === 'PageUp' || e.key === 'Backspace') {
        e.preventDefault();
        prevSlide();
      } else if (e.key === 'Home') {
        e.preventDefault();
        currentSlide = 0;
        updatePresentationUI();
      } else if (e.key === 'End') {
        e.preventDefault();
        currentSlide = totalSlides - 1;
        updatePresentationUI();
      } else if (e.key.toLowerCase() === 'n') {
        drawer.classList.toggle('open');
      } else if (e.key.toLowerCase() === 'f') {
        if (!document.fullscreenElement) {
          document.documentElement.requestFullscreen().catch(() => {});
        } else {
          document.exitFullscreen().catch(() => {});
        }
      }
    });

    /* UI Event Listeners */
    nextBtn.addEventListener('click', nextSlide);
    prevBtn.addEventListener('click', prevSlide);
    drawerClose.addEventListener('click', () => drawer.classList.remove('open'));
    
    prompterLangBtn.addEventListener('click', () => {
      prompterLang = prompterLang === 'en' ? 'hi' : 'en';
      updatePrompter();
    });

    updatePresentationUI();
